  e.preventDefault();

  let valid=true;

  function chk(id,errId,cond){const el=document.getElementById(id);const err=document.getElementById(errId);if(!cond){el.classList.add('is-error');err.classList.add('is-show');valid=false;}else{el.classList.remove('is-error');err.classList.remove('is-show');}}

  chk('q-category','err-q-cat',document.getElementById('q-category').value!=='');

  chk('q-content','err-q-content',document.getElementById('q-content').value.trim().length>0);

  writeForm.classList.remove('is-show');
