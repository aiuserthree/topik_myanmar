  });

  document.getElementById('js-faq-search').value='';

  document.getElementById('js-search-result').style.display='none';

}

document.querySelectorAll('.filter-btn,.aside-cat-btn').forEach(btn=>{

  btn.addEventListener('click',function(){filterFaq(this.dataset.cat)});

});


/* FAQ 검색 */

function searchFaq(){

  const keyword=document.getElementById('js-faq-search').value.trim().toLowerCase();

  const resultEl=document.getElementById('js-search-result');

  if(!keyword){filterFaq('all');resultEl.style.display='none';return}

  document.querySelectorAll('.filter-btn,.aside-cat-btn').forEach(btn=>btn.classList.toggle('is-active',btn.dataset.cat==='all'));

  const items=document.querySelectorAll('.faq-item');

  let count=0;

  items.forEach(item=>{

    const q=item.querySelector('.faq-q-text').textContent.toLowerCase();

    const a=item.querySelector('.faq-item-a').textContent.toLowerCase();

    const match=q.includes(keyword)||a.includes(keyword);

    item.classList.toggle('is-hidden',!match);

    if(match)count++;

  });

  document.querySelectorAll('.faq-section').forEach(section=>{

    const visible=[...section.querySelectorAll('.faq-item')].some(i=>!i.classList.contains('is-hidden'));

    section.style.display=visible?'':'none';

  });

  resultEl.style.display='block';

  resultEl.innerHTML=`<strong>'${keyword}'</strong> 검색 결과 : ${count}건`;

}

document.getElementById('js-faq-search-btn').addEventListener('click',searchFaq);

document.getElementById('js-faq-search').addEventListener('keydown',e=>{if(e.key==='Enter')searchFaq()});

</script>
