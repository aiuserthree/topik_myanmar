<script>

/* ── Mobile Drawer ── */

const hamburger   = document.getElementById('js-hamburger');

const drawer      = document.getElementById('js-drawer');

const overlay     = document.getElementById('js-overlay');

const drawerClose = document.getElementById('js-drawer-close');


function openDrawer() {

  drawer.classList.add('is-open');

  overlay.classList.add('is-show');

  hamburger.setAttribute('aria-expanded', 'true');

  document.body.style.overflow = 'hidden';

}

function closeDrawer() {

  drawer.classList.remove('is-open');

  overlay.classList.remove('is-show');

  hamburger.setAttribute('aria-expanded', 'false');

  document.body.style.overflow = '';

}


hamburger.addEventListener('click', openDrawer);

drawerClose.addEventListener('click', closeDrawer);

overlay.addEventListener('click', closeDrawer);

document.addEventListener('keydown', e => { if (e.key === 'Escape') closeDrawer(); });


/* ── Score bar animate on scroll ── */

const fills = document.querySelectorAll('.score-bar-fill');

const io = new IntersectionObserver(entries => {

  entries.forEach(entry => {

    if (entry.isIntersecting) {

      entry.target.style.width = entry.target.dataset.width || entry.target.style.width;

      io.unobserve(entry.target);

    }

  });

}, { threshold: 0.3 });

fills.forEach(el => io.observe(el));


/* ── Language toggle ── */

function setLang(code) {

  try { localStorage.setItem('tm_lang', code); } catch(e) {}

  document.querySelectorAll('.lang-toggle button, .drawer-lang button').forEach(btn => {

    const match = btn.textContent.trim().toLowerCase().startsWith(code.toLowerCase())

                  || btn.getAttribute('onclick')?.includes(`'${code}'`);

    btn.classList.toggle('is-active', !!match);

    btn.setAttribute('aria-pressed', String(!!match));

  });

}

/* restore saved lang */

try {

  const saved = localStorage.getItem('tm_lang') || 'ko';

  setLang(saved);

} catch(e) {}

</script>
