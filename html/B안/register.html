const photoInput=document.getElementById('photo-input');

const uploadZone=document.getElementById('upload-zone');

const photoPlaceholder=document.getElementById('photo-placeholder');

const photoPreview=document.getElementById('photo-preview');

const photoPreviewImg=document.getElementById('photo-preview-img');

const photoName=document.getElementById('photo-name');

const photoSize=document.getElementById('photo-size');

const summaryPhoto=document.getElementById('summary-photo');

const summaryPhotoName=document.getElementById('summary-photo-name');

let photoFile=null;


photoInput.addEventListener('change',function(){

  const file=this.files[0];

  if(!file)return;

  if(file.size>2*1024*1024){alert('파일 크기가 2MB를 초과합니다.');this.value='';return}

  photoFile=file;

  const reader=new FileReader();

  reader.onload=function(e){

    photoPreviewImg.src=e.target.result;

    summaryPhoto.src=e.target.result;

    summaryPhoto.style.display='block';

    summaryPhotoName.style.display='none';

  };

  reader.readAsDataURL(file);

  photoName.textContent=file.name;

  photoSize.textContent=(file.size/1024).toFixed(1)+' KB';

  photoPlaceholder.style.display='none';

  photoPreview.classList.add('is-show');

  uploadZone.classList.add('has-file');

});

document.getElementById('photo-remove-btn').addEventListener('click',function(e){

  e.stopPropagation();

  photoInput.value='';

  photoFile=null;

  photoPlaceholder.style.display='block';

  photoPreview.classList.remove('is-show');

  uploadZone.classList.remove('has-file');

  summaryPhoto.style.display='none';

  summaryPhotoName.style.display='';

});
