function statusChip(s){

  const level = document.getElementById('levelFilter').value;

  return APPLICANTS.filter(a=>{

    if(currentStatus!=='all' && a.status!==currentStatus) return false;

    if(level && a.level!==level) return false;

    if(q && !(a.nko.includes(q)||a.nen.toLowerCase().includes(q)||a.dob.includes(q)||a.examno.includes(q))) return false;

    return true;

  });

}

function renderPager(){

  const total = Math.ceil(filtered.length/PAGE_SIZE)||1;

  const pager = document.getElementById('pager');

  let html='';

  html+=`<button class="pg-btn" onclick="goPage(${currentPage-1})" ${currentPage===1?'disabled':''}>&#8249;</button>`;

  for(let i=1;i<=total;i++){

    html+=`<button class="pg-btn ${i===currentPage?'on':''}" onclick="goPage(${i})">${i}</button>`;

  }

  html+=`<button class="pg-btn" onclick="goPage(${currentPage+1})" ${currentPage===total?'disabled':''}>&#8250;</button>`;

  pager.innerHTML=html;

function goPage(p){

  renderPager();

function bulkAction(type){

  if(!selectedIds.size){ alert('선택된 항목이 없습니다.'); return; }

  const names=APPLICANTS.filter(a=>selectedIds.has(a.id)).map(a=>a.nko).join(', ');

  if(type==='payment'){

    if(!confirm(`${selectedIds.size}명을 수납 완료 처리하시겠습니까?\n${names}`)) return;

    selectedIds.forEach(id=>{ const a=getById(id); if(a)a.payment='paid'; });

    selectedIds.forEach(id=>{ const a=getById(id); if(a&&a.status==='수납대기')a.status='승인완료'; });

    alert('일괄 수납 처리되었습니다.'); selectedIds.clear(); renderTable();

  } else if(type==='approve'){

    if(!confirm(`${selectedIds.size}명을 승인 처리하시겠습니까?\n${names}`)) return;

    selectedIds.forEach(id=>{ const a=getById(id); if(a){a.status='승인완료';a.history.push({action:'승인완료',by:'manager',at:nowStr()});} });

    alert('일괄 승인 처리되었습니다.'); selectedIds.clear(); renderTable();

  } else if(type==='reject'){

    const reason=prompt('반려 사유를 입력해 주세요');

    if(!reason) return;

    selectedIds.forEach(id=>{ const a=getById(id); if(a){a.status='반려';a.history.push({action:`반려(${reason})`,by:'manager',at:nowStr()});} });

    alert('일괄 반려 처리되었습니다.'); selectedIds.clear(); renderTable();

  if(!confirm('승인완료 상태 접수자에게 수험번호를 일괄 부여하시겠습니까?')) return;

  const start=+(document.getElementById('examNoStart').value)||1;

  const targets=APPLICANTS.filter(a=>a.status==='승인완료'&&!a.examno);

  targets.forEach((a,i)=>{ a.examno=`106-${String(start+i).padStart(3,'0')}`; });

  closeModal('examNoModal');

  renderTable();

  alert(`${targets.length}명에게 수험번호가 부여되었습니다.`);

}

function nowStr(){

  const d=new Date();

  return d.getFullYear()+'.'+(d.getMonth()+1).toString().padStart(2,'0')+'.'+d.getDate().toString().padStart(2,'0')+' '+d.toTimeString().slice(0,5);

}
