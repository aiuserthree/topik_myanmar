        <div style="display:flex;align-items:center;gap:8px;margin-bottom:4px">

          <span class="chip chip-blue" style="font-size:.65rem">${f.cat}</span>

          ${f.vis?'':'<span class="chip chip-gray" style="font-size:.65rem">숨김</span>'}

          <span style="font-size:.7rem;color:var(--gray-400)">순서: ${f.order}</span>

        </div>

        <div class="faq-q">Q. ${f.qKo}</div>

        <div class="faq-a">A. ${f.aKo}</div>

      </div>

      <div style="display:flex;gap:4px;flex-shrink:0">

        <button class="btn btn-outline btn-xs" onclick="openFaqModal(${f.id})">수정</button>

        <button class="btn btn-danger btn-xs" onclick="deleteFaq(${f.id})">삭제</button>

      </div>

    </div>

  `).join('');

}


function openFaqModal(id){

  editFaqId=id;

  const f=id?FAQS.find(x=>x.id===id):null;

  document.getElementById('fm-title').textContent=f?'FAQ 수정':'FAQ 등록';

  document.getElementById('fm-cat').value=f?f.cat:'접수';

  document.getElementById('fm-lang').value=f?f.lang:'전체';

  document.getElementById('fm-vis').value=f?f.vis:1;

  document.getElementById('fm-q-ko').value=f?f.qKo:'';

  document.getElementById('fm-q-my').value=f?f.qMy:'';

  document.getElementById('fm-a-ko').value=f?f.aKo:'';

  document.getElementById('fm-a-my').value=f?f.aMy:'';

  document.getElementById('fm-order').value=f?f.order:'';

  openModal('faqModal');

}

function saveFaq(){

  const qKo=document.getElementById('fm-q-ko').value.trim();

  const aKo=document.getElementById('fm-a-ko').value.trim();

  if(!qKo||!aKo){alert('질문과 답변(한국어)을 입력해 주세요.');return;}

  const data={cat:document.getElementById('fm-cat').value,qKo,qMy:document.getElementById('fm-q-my').value.trim(),aKo,aMy:document.getElementById('fm-a-my').value.trim(),lang:document.getElementById('fm-lang').value,vis:+document.getElementById('fm-vis').value,order:+document.getElementById('fm-order').value||99};

  if(editFaqId){const idx=FAQS.findIndex(f=>f.id===editFaqId);FAQS[idx]={...FAQS[idx],...data};}

  else{FAQS.push({id:Date.now(),...data});}

  closeModal('faqModal');renderFaqs();

}

function deleteFaq(id){if(!confirm('삭제하시겠습니까?'))return;FAQS=FAQS.filter(f=>f.id!==id);renderFaqs();}
