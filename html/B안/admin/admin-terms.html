    <td>${roleChip(a.role)}</td>

    <td style="font-size:.78rem;color:var(--gray-500)">${a.joined}</td>

    <td style="font-size:.78rem;color:var(--gray-500)">${a.lastLogin}</td>

    <td style="text-align:center">${a.failCount>0?`<span style="color:var(--danger);font-weight:700">${a.failCount}회</span>`:'0회'}</td>

    <td>${statusChip(a.status)}</td>

    <td><div style="display:flex;gap:4px;flex-wrap:wrap">

      <button class="btn btn-outline btn-xs" onclick="openAccModal(${a.id})">수정</button>

      ${a.failCount>0?`<button class="btn btn-outline btn-xs" onclick="resetFail(${a.id})">잠금해제</button>`:''}

      <button class="btn btn-outline btn-xs" onclick="resetPw(${a.id})">비밀번호초기화</button>

      <button class="btn btn-danger btn-xs" onclick="deleteAcc(${a.id})">삭제</button>

    </div></td>

  </tr>`).join('');

}


function openAccModal(id){

  editAccId=id;

  const a=id?ACCOUNTS.find(x=>x.id===id):null;

  document.getElementById('am-title').textContent=a?'계정 수정':'계정 등록';

  document.getElementById('am-id').value=a?a.loginId:'';

  document.getElementById('am-id').disabled=!!a;

  document.getElementById('am-name').value=a?a.name:'';

  document.getElementById('am-email').value=a?a.email:'';

  document.getElementById('am-pw').value='';

  document.getElementById('am-role').value=a?a.role:'일반관리자';

  document.getElementById('am-status').value=a?a.status:'활성';

  openModal('accModal');

}

function saveAcc(){

  const loginId=document.getElementById('am-id').value.trim();

  const name=document.getElementById('am-name').value.trim();

  const email=document.getElementById('am-email').value.trim();

  if(!loginId||!name||!email){alert('아이디, 성명, 이메일은 필수입니다.');return;}

  if(!editAccId&&!document.getElementById('am-pw').value){alert('신규 등록 시 비밀번호는 필수입니다.');return;}

  const data={loginId,name,email,role:document.getElementById('am-role').value,status:document.getElementById('am-status').value,failCount:0};

  if(editAccId){const idx=ACCOUNTS.findIndex(a=>a.id===editAccId);ACCOUNTS[idx]={...ACCOUNTS[idx],...data};}

  else{ACCOUNTS.push({id:Date.now(),joined:new Date().toLocaleDateString('ko-KR').replace(/\. /g,'.').replace('.','.').slice(0,-1),lastLogin:'—',...data});}

  closeModal('accModal');renderAccounts();

}

function resetFail(id){const a=ACCOUNTS.find(x=>x.id===id);if(a&&confirm('로그인 잠금을 해제하시겠습니까?')){a.failCount=0;renderAccounts();}}

function resetPw(id){const a=ACCOUNTS.find(x=>x.id===id);if(a&&confirm(`${a.name}의 비밀번호를 초기화하시겠습니까? 임시 비밀번호가 이메일로 발송됩니다.`))alert('임시 비밀번호가 이메일로 발송되었습니다.');}

function deleteAcc(id){const a=ACCOUNTS.find(x=>x.id===id);if(a&&confirm(`${a.name} 계정을 삭제하시겠습니까?`)){ACCOUNTS=ACCOUNTS.filter(x=>x.id!==id);renderAccounts();}}

function togglePw(id){const el=document.getElementById(id);el.type=el.type==='password'?'text':'password';}

function openModal(id){document.getElementById(id).classList.add('open');}

function closeModal(id){document.getElementById(id).classList.remove('open');}

document.querySelectorAll('.overlay').forEach(ov=>{ov.addEventListener('click',function(e){if(e.target===this)closeModal(this.id);});});
