let currentIqId=null;

let iqStatus='all';


function statusChip(s){

  return s==='답변완료'

    ? '<span class="chip chip-green">답변완료</span>'

    : '<span class="chip chip-yellow">답변대기</span>';

}

function catChip(c){return `<span class="chip chip-blue" style="font-size:.65rem">${c}</span>`;}


function getFiltered(){

  const cat=document.getElementById('iqCat').value;

  const q=document.getElementById('iqSearch').value.trim().toLowerCase();

  return INQUIRIES.filter(i=>{

    if(iqStatus!=='all'&&i.status!==iqStatus)return false;

    if(cat&&i.cat!==cat)return false;

    if(q&&!(i.title.toLowerCase().includes(q)||i.author.includes(q)))return false;

    return true;

  });

}


function renderInquiries(){

  const list=getFiltered();

  document.getElementById('iqTbody').innerHTML=list.length

    ? list.map(i=>`<tr>

        <td style="font-weight:600">${i.id}</td>

        <td>${catChip(i.cat)}</td>

        <td style="max-width:260px;white-space:nowrap;overflow:hidden;text-overflow:ellipsis;font-weight:500">${i.priv?'🔒 ':''}<a href="#" onclick="openIqDetail(${i.id});return false">${i.title}</a></td>

        <td>${i.author}</td>

        <td style="font-size:.78rem;color:var(--gray-500)">${i.date}</td>

        <td>${statusChip(i.status)}</td>

        <td><button class="btn btn-primary btn-xs" onclick="openIqDetail(${i.id})">상세·답변</button></td>

      </tr>`).join('')

    : `<tr><td colspan="7" style="text-align:center;padding:40px;color:var(--gray-400)">조회된 문의가 없습니다.</td></tr>`;


  const counts={};

  INQUIRIES.forEach(i=>{counts[i.status]=(counts[i.status]||0)+1;});

  document.querySelectorAll('.iqcnt').forEach(el=>{

    const s=el.dataset.s;

    el.textContent=s==='all'?INQUIRIES.length:(counts[s]||0);

  });

}


document.getElementById('iqFilters').querySelectorAll('.filter-chip').forEach(btn=>{

    this.classList.add('on');

    iqStatus=this.dataset.s;

    renderInquiries();

  });

});


function resetIqFilter(){

  iqStatus='all';

  document.querySelectorAll('#iqFilters .filter-chip').forEach(b=>b.classList.remove('on'));

  document.querySelector('#iqFilters .filter-chip[data-s="all"]').classList.add('on');

  document.getElementById('iqCat').value='';

  document.getElementById('iqSearch').value='';

  renderInquiries();

}


function openIqDetail(id){

  currentIqId=id;

  const i=INQUIRIES.find(x=>x.id===id);

  if(!i)return;

  document.getElementById('iq-modal-title').textContent=`문의 상세 — ${i.title.slice(0,20)}`;

  document.getElementById('iq-cat').textContent=i.cat;

  document.getElementById('iq-author').textContent=i.author;

  document.getElementById('iq-contact').textContent=i.contact;

  document.getElementById('iq-date').textContent=i.date;

  document.getElementById('iq-priv').textContent=i.priv?'비공개':'공개';

  document.getElementById('iq-content').textContent=i.content;

  document.getElementById('iq-reply').value='';

  const prevArea=document.getElementById('iq-prev-reply-area');

  if(i.reply){

    prevArea.style.display='block';

    document.getElementById('iq-prev-reply').textContent=i.reply;

  } else {

    prevArea.style.display='none';

  }

  openModal('iqModal');

}


function doIqReply(){

  const i=INQUIRIES.find(x=>x.id===currentIqId);

  if(!i)return;

  const reply=document.getElementById('iq-reply').value.trim();

  if(!reply){alert('답변 내용을 입력해 주세요.');return;}

  i.reply=reply;

  i.status='답변완료';

  closeModal('iqModal');

  renderInquiries();

  alert('답변이 저장되었습니다.');

}


function deleteInquiry(){

  if(!confirm('이 문의를 삭제하시겠습니까?'))return;

  INQUIRIES=INQUIRIES.filter(i=>i.id!==currentIqId);

  closeModal('iqModal');

  renderInquiries();

}
