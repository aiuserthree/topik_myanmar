  const tbody=document.getElementById('venueTbody');

  if(!VENUES.length){tbody.innerHTML=`<tr><td colspan="8" style="text-align:center;padding:40px;color:var(--gray-400)">등록된 시험장이 없습니다.</td></tr>`;return;}

  tbody.innerHTML=VENUES.map(v=>{

    const pct=Math.min(100,Math.round(70/v.cap*100));

    return `<tr>

      <td style="font-weight:600">${v.name}</td>

      <td>${v.city}</td>

      <td style="font-size:.78rem;color:var(--gray-500);max-width:180px">${v.addr||'—'}</td>

      <td style="font-size:.78rem">${v.tel||'—'}</td>

      <td>${v.cap}</td>

      <td style="font-size:.78rem;color:var(--gray-600)">${v.rounds||'—'}</td>

      <td>${v.active?'<span class="chip chip-green">사용</span>':'<span class="chip chip-gray">미사용</span>'}</td>

      <td><div style="display:flex;gap:4px"><button class="btn btn-outline btn-xs" onclick="openVenueModal(${v.id})">수정</button><button class="btn btn-danger btn-xs" onclick="openDelModal(${v.id})">삭제</button></div></td>

    </tr>`;

  }).join('');

}


function openVenueModal(id){

  editVenueId=id;

  const v=id?VENUES.find(x=>x.id===id):null;

  document.getElementById('vm-title').textContent=v?'시험장 수정':'시험장 등록';

  document.getElementById('vm-name').value=v?v.name:'';

  document.getElementById('vm-city').value=v?v.city:'';

  document.getElementById('vm-cap').value=v?v.cap:'';

  document.getElementById('vm-addr').value=v?v.addr:'';

  document.getElementById('vm-tel').value=v?v.tel:'';

  document.getElementById('vm-manager').value=v?v.manager:'';

  document.getElementById('vm-direction').value=v?v.direction:'';

  document.getElementById('vm-note').value=v?v.note:'';

  document.getElementById('vm-active').value=v?v.active:1;

  openModal('venueModal');

}

function saveVenue(){

  const name=document.getElementById('vm-name').value.trim();

  const city=document.getElementById('vm-city').value.trim();

  if(!name||!city){alert('시험장명과 도시/지역은 필수입니다.');return;}

  const data={name,city,cap:+document.getElementById('vm-cap').value||0,addr:document.getElementById('vm-addr').value.trim(),tel:document.getElementById('vm-tel').value.trim(),manager:document.getElementById('vm-manager').value.trim(),direction:document.getElementById('vm-direction').value.trim(),note:document.getElementById('vm-note').value.trim(),active:+document.getElementById('vm-active').value,rounds:'—'};

  if(editVenueId){const idx=VENUES.findIndex(v=>v.id===editVenueId);VENUES[idx]={...VENUES[idx],...data};}

  else{VENUES.push({id:Date.now(),...data});}

  closeModal('venueModal');renderVenues();

}

function openDelModal(id){delVenueId=id;const v=VENUES.find(x=>x.id===id);document.getElementById('del-name').textContent=v?v.name:'—';openModal('delModal');}

function doDelVenue(){VENUES=VENUES.filter(v=>v.id!==delVenueId);closeModal('delModal');renderVenues();}

function openModal(id){document.getElementById(id).classList.add('open');}

function closeModal(id){document.getElementById(id).classList.remove('open');}

document.querySelectorAll('.overlay').forEach(ov=>{ov.addEventListener('click',function(e){if(e.target===this)closeModal(this.id);});});

function doLogout(){sessionStorage.clear();location.href='admin-login.html';}

(function(){

  const admin=JSON.parse(sessionStorage.getItem('bo_admin')||'{"name":"관리자","role":"최고관리자"}');

  const ini=(admin.name||'관')[0];

  ['sb-av','tb-av'].forEach(id=>document.getElementById(id).textContent=ini);

  document.getElementById('sb-name').textContent=admin.name;document.getElementById('sb-role').textContent=admin.role;

  document.getElementById('tb-name').textContent=admin.name;document.getElementById('tb-role').textContent=admin.role;

  const sidebar=document.getElementById('sidebar');const backdrop=document.getElementById('backdrop');

  document.getElementById('hbBtn').addEventListener('click',()=>{const o=sidebar.classList.toggle('open');backdrop.classList.toggle('show',o);});

  backdrop.addEventListener('click',()=>{sidebar.classList.remove('open');backdrop.classList.remove('show');});

  document.getElementById('logoutBtn').addEventListener('click',()=>openModal('logoutModal'));

  renderVenues();

})();
