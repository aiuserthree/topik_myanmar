/* ── STATUS CHIP ── */

function statusChip(s){

  const m={대기:'chip-yellow',처리중:'chip-blue',완료:'chip-green',거절:'chip-red'};

  return `<span class="chip ${m[s]||'chip-gray'}">${s}</span>`;

}

function typeChip(t){

  return t==='환불'

    ? `<span class="chip chip-red">${t}</span>`

    : `<span class="chip chip-purple">${t}</span>`;

}


/* ── RENDER ── */

function getFiltered(){

  const type = document.getElementById('rfType').value;

  const round = document.getElementById('rfRound').value;

  const q = document.getElementById('rfSearch').value.trim().toLowerCase();

  return REFUNDS.filter(r=>{

    if(currentStatus !== 'all' && r.status !== currentStatus) return false;

    if(type && r.type !== type) return false;

    if(round && r.round !== round) return false;

    if(q && !(r.name.includes(q) || r.content.toLowerCase().includes(q))) return false;

    return true;

  });

}


function renderRefunds(){

  const list = getFiltered();

  document.getElementById('rfTbody').innerHTML = list.length

    ? list.map(r=>`
