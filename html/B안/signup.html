/* ── Step logic ── */

let currentStep=1;

function goStep(n){

  document.querySelectorAll('.step-panel').forEach(p=>p.classList.remove('is-active'));

  document.getElementById('panel-'+n).classList.add('is-active');

  [1,2,3].forEach(i=>{

    const sc=document.getElementById('sc-'+i);

    const sl=document.getElementById('sl-'+i);

    sc.className='step-circle '+(i<n?'done':(i===n?'active':'todo'));

    if(i<n)sc.textContent='✓';

    else sc.textContent=i;

    sl.className='step-label'+(i===n?' active':'');

  });

  document.querySelector('.step-item[data-step="'+Math.max(n-1,1)+'"]').classList.add('is-done');

  if(n===3)document.getElementById('js-auth-foot').style.display='none';

  window.scrollTo({top:0,behavior:'smooth'});

  currentStep=n;

}


/* ── Step 1: 약관 ── */

const agreeAll=document.getElementById('agree-all');

const agreeItems=document.querySelectorAll('.agree-item');

agreeAll.addEventListener('change',function(){

  agreeItems.forEach(cb=>cb.checked=this.checked);

});

agreeItems.forEach(cb=>{

  cb.addEventListener('change',()=>{

    agreeAll.checked=[...agreeItems].every(c=>c.checked);

  });

});

document.getElementById('btn-step1-next').addEventListener('click',function(){

  const required=['terms','privacy'];

  const missing=required.some(t=>!document.querySelector('.agree-item[data-target="'+t+'"]').checked);

  const errEl=document.getElementById('err-agree');

  if(missing){errEl.style.display='block';return}

  errEl.style.display='none';

  goStep(2);

});

/* ── PW strength ── */

document.getElementById('pw').addEventListener('input',function(){

  const v=this.value;

  const segs=[document.getElementById('seg1'),document.getElementById('seg2'),document.getElementById('seg3'),document.getElementById('seg4')];

  const label=document.getElementById('pw-label');

  const hasLen=v.length>=8;

  const hasUpper=/[A-Z]/.test(v);

  const hasNum=/[0-9]/.test(v);

  const hasSp=/[^A-Za-z0-9]/.test(v);

  const score=[hasLen,hasUpper||hasNum,hasNum&&hasUpper,hasSp].filter(Boolean).length;

  const cls=['','weak','fair','good','strong'];

  const txt=['','매우 약함','보통','강함','매우 강함'];

  segs.forEach((s,i)=>{s.className='pw-seg'+(i<score?' '+cls[score]:'')});

  label.textContent=v?txt[score]:'영문+숫자+특수문자 조합 8자 이상';

  /* pw confirm live check */

  const confirm=document.getElementById('pw-confirm');

  if(confirm.value)checkPwMatch();

});

function checkPwMatch(){

  const pw=document.getElementById('pw').value;

  const pc=document.getElementById('pw-confirm').value;

  const err=document.getElementById('err-pw-confirm');

  const ok=document.getElementById('ok-pw-confirm');

  if(!pc)return;

  if(pw===pc){err.classList.remove('is-show');ok.classList.add('is-show');document.getElementById('pw-confirm').classList.add('is-ok');document.getElementById('pw-confirm').classList.remove('is-error')}

  else{ok.classList.remove('is-show');err.classList.add('is-show');document.getElementById('pw-confirm').classList.add('is-error');document.getElementById('pw-confirm').classList.remove('is-ok')}

}

document.getElementById('pw-confirm').addEventListener('input',checkPwMatch);


/* ── Step 2: 제출 ── */

document.getElementById('js-signup-form').addEventListener('submit',function(e){

  e.preventDefault();

  let valid=true;

  function chk(id,errId,cond){

    const el=document.getElementById(id);

    const err=document.getElementById(errId);

    if(!cond){el.classList.add('is-error');err.classList.add('is-show');valid=false;}

    else{el.classList.remove('is-error');err.classList.remove('is-show');}

  }

  const nameEn=document.getElementById('name-en').value.trim();

  const birth=document.getElementById('birth').value;

  const gender=document.getElementById('gender').value;

  const nationality=document.getElementById('nationality').value;

  const phone=document.getElementById('phone').value.trim();

  const email=document.getElementById('email').value.trim();

  const pw=document.getElementById('pw').value;

  const pwc=document.getElementById('pw-confirm').value;

  chk('name-en','err-name-en',nameEn.length>0);

  chk('birth','err-birth',birth!=='');

  chk('gender','err-gender',gender!=='');

  chk('nationality','err-nationality',nationality!=='');

  chk('phone','err-phone',phone.length>5);

  chk('email','err-email',/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email));

  chk('pw','err-pw',pw.length>=8);

  chk('pw-confirm','err-pw-confirm',pw===pwc&&pwc.length>0);

  if(valid)goStep(3);

});
