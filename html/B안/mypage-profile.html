function toggleEdit(on){

  isEditing=on;

  viewMode.style.display=on?'none':'block';

  editForm.style.display=on?'block':'none';

  btnEditToggle.textContent=on?'취소':'수정';

  e.preventDefault();

  let valid=true;

  function chk(id,errId,cond){

    const el=document.getElementById(id);const err=document.getElementById(errId);

    if(!cond){el.classList.add('is-error');err.classList.add('is-show');valid=false;}

    else{el.classList.remove('is-error');err.classList.remove('is-show');}

  }

  const nameEn=document.getElementById('edit-name-en').value.trim();

  const phone=document.getElementById('edit-phone').value.trim();

  const email=document.getElementById('edit-email').value.trim();

  chk('edit-name-en','err-edit-name',nameEn.length>0);

  chk('edit-phone','err-edit-phone',phone.length>4);

  chk('edit-email','err-edit-email',/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email));

  if(!valid)return;

  /* 저장 후 뷰 업데이트 */

  document.getElementById('view-name-en').textContent=nameEn;

  document.getElementById('view-name-ko').textContent=document.getElementById('edit-name-ko').value||'-';

  toggleEdit(false);

  const alert=document.getElementById('info-save-alert');

  alert.classList.add('is-show');

  setTimeout(()=>alert.classList.remove('is-show'),4000);

});

  else{document.getElementById('new-pw').classList.remove('is-error');errNw.classList.remove('is-show')}

  if(nw!==nc||!nc){document.getElementById('new-pw-confirm').classList.add('is-error');errNc.classList.add('is-show');valid=false;}

  else{document.getElementById('new-pw-confirm').classList.remove('is-error');errNc.classList.remove('is-show')}

  if(!valid)return;

  const ok=document.getElementById('pw-save-alert');

  ok.classList.add('is-show');

  this.reset();

  setTimeout(()=>ok.classList.remove('is-show'),4000);

});
