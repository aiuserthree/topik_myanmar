#### 4.3 할인 구간 추가 모달
```javascript
function saveDiscountAdd() {
    const minAmount = document.getElementById('minAmount').value;
    const maxAmount = document.getElementById('maxAmount').value;
    const discountRate = document.getElementById('discountRate').value;
    
    // 유효성 검사
    if (!minAmount || minAmount <= 0) {
        showToast('최소 금액을 입력해주세요.', 'error');
        return;
    }
    
    if (!discountRate || discountRate <= 0 || discountRate > 100) {
        showToast('할인율은 1~100 사이의 값이어야 합니다.', 'error');
        return;
    }
    
    // 저장 처리
    showToast('할인 구간이 추가되었습니다.', 'success');
    closeModal('addDiscountModal');
}
```
