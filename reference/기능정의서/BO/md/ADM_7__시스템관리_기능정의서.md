// IP 화이트리스트 추가
function addIpWhitelist() {
    const ip = document.getElementById('newIpAddress').value;
    
    // IP 형식 검증
    const ipRegex = /^(\d{1,3}\.){3}\d{1,3}$/;
    if (!ipRegex.test(ip)) {
        showToast('올바른 IP 주소 형식이 아닙니다.', 'error');
        return;
    }
    
    showToast('IP가 화이트리스트에 추가되었습니다.', 'success');
}
